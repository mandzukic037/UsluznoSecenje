-- Kreirati bazu:
CREATE DATABASE IF NOT EXISTS usluznosecenje
  CHARACTER SET utf8mb4
  COLLATE utf8mb4_unicode_ci;

USE usluznosecenje;

-- Tabela: products
CREATE TABLE IF NOT EXISTS products (
    id          BIGINT AUTO_INCREMENT PRIMARY KEY,
    title       VARCHAR(255)  NOT NULL,
    category    VARCHAR(100),
    price       DOUBLE        NOT NULL DEFAULT 0,
    image       VARCHAR(500),
    description VARCHAR(1000),
    material    VARCHAR(255),
    thickness   VARCHAR(100)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Tabela: porudzbine
CREATE TABLE IF NOT EXISTS porudzbine (
    id              BIGINT AUTO_INCREMENT PRIMARY KEY,
    ime             VARCHAR(100),
    prezime         VARCHAR(100),
    email           VARCHAR(255),
    telefon         VARCHAR(50),
    firma           VARCHAR(255),
    pib             VARCHAR(100),
    adresa          VARCHAR(255),
    grad            VARCHAR(100),
    post_broj       VARCHAR(20),
    nacin_dostave   VARCHAR(100),
    placanje        VARCHAR(100),
    napomena        VARCHAR(1000),
    ukupno_bez_pdv  DOUBLE DEFAULT 0,
    pdv             DOUBLE DEFAULT 0,
    ukupno_sa_pdv   DOUBLE DEFAULT 0,
    status          ENUM('NOVA','U_OBRADI','POSLATA','OTKAZANA') DEFAULT 'NOVA',
    kreirano_u      DATETIME
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- Tabela: porudzbine_stavke
CREATE TABLE IF NOT EXISTS porudzbine_stavke (
    id              BIGINT AUTO_INCREMENT PRIMARY KEY,
    porudzbina_id   BIGINT NOT NULL,
    product_id      BIGINT,
    naziv           VARCHAR(255),
    kolicina        INT DEFAULT 1,
    cena            DOUBLE DEFAULT 0,
    CONSTRAINT fk_stavka_porudzbina
        FOREIGN KEY (porudzbina_id)
        REFERENCES porudzbine(id)
        ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
